const unit=(price,ch,cost,ship,op)=>{
  const c=price*ch.rate+ch.fixed;
  const t=price*(v('taxRate')+v('anticipRate'))/100;
  const a=price*v('adsPercent')/100;
  return {c, t, a, p: price - cost - c - t - a - ship - op};
};

function calcPricing(){
  const ch=CH[$('calcChannel').value], cost=v('costPrice'), price=v('salePrice'), ship=v('shippingCost'), op=v('operationalCost');
  const u=unit(price, ch, cost, ship, op);
  G.profit=u.p; G.margin=price?u.p/price*100:0; G.preAds=price?(u.p+u.a)/price*100:0;
  set('netProfitVal', R(u.p));
  $('netProfitVal').className='text-3xl font-extrabold tracking-tight '+(u.p>=0?'text-emerald-600':'text-rose-600');
  set('netMarginVal', G.margin.toFixed(1)+'%');
  set('markupVal', (cost?price/cost:0).toFixed(2)+'x');
  set('commVal', R(u.c));
  set('taxVal', R(u.t));
  set('roiVal', (cost?u.p/cost*100:0).toFixed(1)+'%');
  const d=1 - ch.rate - (v('taxRate')+v('anticipRate')+v('adsPercent')+v('targetMargin'))/100;
  set('sug20', d>0 ? R((cost + ship + op + ch.fixed)/d) : 'Inviável');
}

function calcKit(){
  const ch=CH[$('calcChannel').value], ship=v('shippingCost'), op=v('operationalCost'), price=v('kitSalePrice');
  const cost=kit.reduce((a,k)=>a+k.cost*k.qty,0), units=kit.reduce((a,k)=>a+k.qty,0), u=unit(price, ch, cost, ship, op);
  set('kitCost', R(cost));
  set('kitSave', R(Math.max(0, units-1)*(ch.fixed + ship + op)));
  set('kitProfit', R(u.p));
  set('kitMargin', (price?u.p/price*100:0).toFixed(1)+'%');
}

function calcChannels(){
  const price=v('salePrice'), cost=v('costPrice'), ship=v('shippingCost'), op=v('operationalCost');
  const rows=[], names=[], vals=[];
  for(const k in CH){
    const c=CH[k], u=unit(price, c, cost, ship, op), m=price?u.p/price*100:0;
    const dd=1 - c.rate - (v('taxRate')+v('anticipRate')+v('adsPercent')+v('targetMargin'))/100;
    names.push(c.name); vals.push(+u.p.toFixed(2));
    rows.push(`<tr><td class="p-3 font-semibold">${c.name}</td><td class="p-3 text-slate-500">${+(c.rate*100).toFixed(1)}% + ${R(c.fixed)}</td><td class="p-3">${R(u.c)}</td><td class="p-3 font-bold ${u.p>=0?'text-emerald-600':'text-rose-500'}">${R(u.p)}</td><td class="p-3 font-bold">${m.toFixed(1)}%</td><td class="p-3">${dd>0?R((cost+ship+op+c.fixed)/dd):'Inviável'}</td></tr>`);
  }
  $('chBody').innerHTML=rows.join('');
  if(!chart){
    chart=new Chart($('chChart'),{type:'bar',data:{labels:names,datasets:[{label:'Lucro (R$)',data:vals,backgroundColor:'rgba(37,99,235,.8)',borderRadius:8}]},options:{responsive:true,maintainAspectRatio:false,plugins:{legend:{display:false}}}});
  } else {
    chart.data.labels=names; chart.data.datasets[0].data=vals; chart.update();
  }
}

function calcStock(){
  const d=v('dailySales'), s=v('currentStock'), lt=v('leadTime');
  const days=d>0?s/d:0, rp=d*lt;
  set('stockDays', d>0?Math.floor(days)+' dias':'-');
  set('locked', R(s*v('costPrice')));
  set('reorder', Math.ceil(rp)+' un');
  G.low=s<=rp;
  set('reorderNote', s<=rp?'Estoque abaixo do ponto de reposição.':'Estoque em nível seguro.');
  const c=days<=30?'A':days<=60?'B':'C';
  set('abc','Classe '+c);
  set('abcNote', c==='A'?'Giro alto (até 30 dias).':c==='B'?'Giro médio (31 a 60 dias).':'Giro baixo (> 60 dias): capital parado.');
}

function calcGoal(){
  const p=G.profit, fx=v('fixedCosts'), tg=v('targetProfit'), pr=v('salePrice');
  if(!(p>0)){
    ['beUnits','tgUnits'].forEach(i=>set(i,'Inviável'));
    ['beRev','tgRev'].forEach(i=>set(i,'Lucro unitário é zero ou negativo.'));
    set('tgDaily',''); G.beOk=false; return;
  }
  const be=Math.ceil(fx/p), tu=Math.ceil((fx+tg)/p);
  G.beOk=true;
  set('beUnits', be+' un'); set('beRev', 'Faturamento: '+R(be*pr));
  set('tgUnits', tu+' un'); set('tgRev', 'Faturamento: '+R(tu*pr));
  set('tgDaily', 'Meta diária necessária: cerca de '+Math.ceil(tu/30)+' vendas/dia');
}

function importCSV(inp){
  const fl=inp.files[0]; if(!fl)return;
  fl.text().then(t=>{
    const L=t.replace(/^\ufeff/,'').split(/\r?\n/).filter(x=>x.trim());
    const sep=L[0].includes(';')?';':',', H=L[0].split(sep).map(x=>x.toLowerCase().trim());
    const ix=re=>H.findIndex(h=>re.test(h));
    const c={n:ix(/produto|t[ií]tulo|nome|sku|item/), q:ix(/quant|qtd|unid/), r:ix(/valor|receita|total|pre[cç]o/), f:ix(/tarifa|comiss|taxa/), s:ix(/frete|envio/), k:ix(/custo/)};
    inp.value='';
    if(c.n<0||c.q<0||c.r<0){set('salesMsg','Colunas não encontradas no CSV.'); return;}
    const num=x=>{let z=String(x||'').replace(/[^\d,.-]/g,''); if(z.includes(','))z=z.replace(/\./g,'').replace(',','.'); return parseFloat(z)||0}, m={};
    L.slice(1).forEach(l=>{const a=l.split(sep).map(x=>x.replace(/^"|"$/g,'').trim()), n=a[c.n]; if(!n)return; const o=m[n.toLowerCase()]??={n,q:0,rev:0,fee:0,ship:0}; o.q+=num(a[c.q]); o.rev+=num(a[c.r]); if(c.f>=0)o.fee+=num(a[c.f]); if(c.s>=0)o.ship+=num(a[c.s]); if(c.k>=0&&num(a[c.k]))o.uc=num(a[c.k])});
    sales=Object.values(m); set('salesMsg',''); calcAll(); save();
  });
}

  const fx=v('fixedCosts'), mo=G.monthly, vd=$('dVerdict');

  const bad='bg-rose-50 dark:bg-rose-950/30 text-rose-800 dark:text-rose-200 border border-rose-200 dark:border-rose-900';

  if(G.real){
    const r=G.real;
    vd.className='mt-4 p-5 rounded-2xl text-base font-bold '+ (r.pr>=0?ok:bad);
    vd.textContent=`Nas vendas importadas você faturou ${R(r.rev)} e ${r.pr>=0?'lucrou':'teve prejuízo de'} ${R(Math.abs(r.pr))} (${r.rev?(r.pr/r.rev*100).toFixed(1):0}%). ${r.loss} produto(s) no vermelho.`;
  } else if(skus.some(k=>k.q>0)){

    vd.className='mt-4 p-5 rounded-2xl text-base font-bold '+(net>=0?ok:bad);

  } else {
    vd.className='mt-4 p-5 rounded-2xl text-base font-bold '+(G.profit>=0?ok:bad);
    vd.textContent=G.profit>=0?`Cada venda deixa ${R(G.profit)} de lucro (${G.margin.toFixed(1)}%).`:`Cada venda dá prejuízo de ${R(-G.profit)}. Ajuste preços ou custos.`;
  }

  const A=[], g=v('payout')-v('supplier');

  if(!A.length) A.push(['g','Nenhum alerta crítico. Operação saudável.']);

  const C={
    r:'bg-rose-50 dark:bg-rose-950/30 text-rose-800 dark:text-rose-200 border border-rose-200 dark:border-rose-900',
    a:'bg-amber-50 dark:bg-amber-950/30 text-amber-800 dark:text-amber-200 border border-amber-200 dark:border-amber-900',

  };
  $('alerts').innerHTML=A.map(a=>`<li class="p-3.5 rounded-xl text-xs font-semibold ${C[a[0]]}">${a[1]}</li>`).join('');
}
